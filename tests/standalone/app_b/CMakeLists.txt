cmake_minimum_required(VERSION 3.16)
project(MadCryptoAppB CXX)
find_package(MadCrypto CONFIG REQUIRED)
add_executable(app_b main.cpp)
target_compile_features(app_b PRIVATE cxx_std_11)
target_compile_options(app_b PRIVATE -Wall -Wextra -Werror)
target_link_libraries(app_b PRIVATE MadCrypto::MadCrypto)
enable_testing()
add_test(NAME independent_app_b COMMAND app_b)
